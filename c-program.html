Topological Sort in C Programming

Production-grade, fully commented C source code demonstrating Kahn's in-degree reduction and DFS recursion with memory breakdown and an interactive terminal runner.

#include <stdio.h> #define MAX 100 int main() { int n, e; int graph[MAX][MAX] = {0}; int indegree[MAX] = {0}; int queue[MAX], front = 0, rear = 0; int result[MAX], count = 0; printf("Enter number of vertices: "); scanf("%d", &n); printf("Enter number of edges: "); scanf("%d", &e); printf("Enter edges (u v):\n"); for (int i = 0; i < e; i++) { int u, v; scanf("%d %d", &u, &v); graph[u][v] = 1; indegree[v]++; } /* Step 1: Find all vertices with in-degree 0 and push to queue */ for (int i = 0; i < n; i++) { if (indegree[i] == 0) { queue[rear++] = i; } } /* Step 2: Process vertices using FIFO Queue */ while (front < rear) { int u = queue[front++]; result[count++] = u; for (int v = 0; v < n; v++) { if (graph[u][v] == 1) { indegree[v]--; if (indegree[v] == 0) { queue[rear++] = v; } } } } /* Step 3: Check for cycles */ if (count != n) { printf("Cycle detected! Topological sorting is not possible.\n"); } else { printf("Topological Order: "); for (int i = 0; i < count; i++) { printf("%d ", result[i]); } printf("\n"); } return 0; }
#include <stdio.h> #include <stdlib.h> struct Node { int dest; struct Node* next; }; struct Graph { int V; struct Node** adjLists; int* indegree; }; struct Node* createNode(int v) { struct Node* newNode = malloc(sizeof(struct Node)); newNode->dest = v; newNode->next = NULL; return newNode; } void addEdge(struct Graph* graph, int src, int dest) { struct Node* newNode = createNode(dest); newNode->next = graph->adjLists[src]; graph->adjLists[src] = newNode; graph->indegree[dest]++; } void topologicalSort(struct Graph* graph) { int* queue = malloc(graph->V * sizeof(int)); int* result = malloc(graph->V * sizeof(int)); int front = 0, rear = 0, count = 0; for (int i = 0; i < graph->V; i++) { if (graph->indegree[i] == 0) { queue[rear++] = i; } } while (front < rear) { int u = queue[front++]; result[count++] = u; struct Node* temp = graph->adjLists[u]; while (temp) { int v = temp->dest; graph->indegree[v]--; if (graph->indegree[v] == 0) { queue[rear++] = v; } temp = temp->next; } } if (count != graph->V) { printf("Cycle detected!\n"); } else { printf("Topological Order: "); for (int i = 0; i < count; i++) { printf("%d ", result[i]); } printf("\n"); } free(queue); free(result); }
#include <stdio.h> #include <stdbool.h> #define MAX 100 int graph[MAX][MAX] = {0}; int visited[MAX] = {0}; // 0: unvisited, 1: visiting, 2: visited int stack[MAX], top = -1; bool hasCycle = false; void push(int v) { stack[++top] = v; } void dfs(int u, int n) { if (hasCycle) return; visited[u] = 1; // Visiting (in call stack) for (int v = 0; v < n; v++) { if (graph[u][v]) { if (visited[v] == 1) { hasCycle = true; // Back-edge found! return; } if (visited[v] == 0) { dfs(v, n); } } } visited[u] = 2; // Fully processed push(u); // Post-order push } int main() { int n = 6; // Sample DAG edges graph[0][1] = 1; graph[0][2] = 1; graph[1][3] = 1; graph[2][3] = 1; graph[3][4] = 1; graph[4][5] = 1; for (int i = 0; i < n; i++) { if (visited[i] == 0) { dfs(i, n); } } if (hasCycle) { printf("Cycle detected! Topological sort impossible.\n"); } else { printf("Topological Order: "); while (top >= 0) { printf("%d ", stack[top--]); } printf("\n"); } return 0; }

🧠 C Variable & Data Structure Breakdown

Understand how each array and pointer in the C implementation manages memory:

graph[MAX][MAX]

2D integer matrix where graph[u][v] = 1 indicates a directed edge from vertex u to v.

indegree[MAX]

1D array tracking unresolved prerequisite dependencies for every vertex.

queue[MAX], front, rear

Linear array-based FIFO queue storing vertices currently ready with in-degree 0.

result[MAX], count

Output array holding the finalized linear topological ordering.
